cmake_minimum_required(VERSION 3.16)
project(MATRYOSHKA_ENGINE LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
add_library(matryoshka_engine SHARED src/mw_engine.c src/mw_scene.cpp)
target_include_directories(matryoshka_engine PUBLIC include)
target_compile_definitions(matryoshka_engine PRIVATE MW_BUILD)
add_executable(engine_demo src/demo.cpp)
target_link_libraries(engine_demo PRIVATE matryoshka_engine)
